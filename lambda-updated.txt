const express = require('express');
const serverless = require('serverless-http');
const cors = require('cors');
const mysql = require('mysql2/promise');

const app = express();
app.use(cors());
app.use(express.json());

const pool = mysql.createPool({
  host: process.env.DB_HOST,
  user: process.env.DB_USER,
  password: process.env.DB_PASSWORD,
  database: process.env.DB_NAME,
  waitForConnections: true,
  connectionLimit: 10,
  ssl: { rejectUnauthorized: false },
  multipleStatements: true
});

const query = async (sql, params) => {
  const [rows] = await pool.execute(sql, params);
  return rows;
};

// 1. Registrar un nuevo Hotel
app.post('/api/hotels', async (req, res) => {
  try {
    const { Nombre, Direccion, Ciudad, Pais, Telefono, EmailContacto, CategoriaEstrellas, DescripcionGeneral } = req.body;

    if (!Nombre || !Direccion) {
      return res.status(400).json({ error: 'Nombre y Dirección son obligatorios' });
    }

    const result = await query(
      `INSERT INTO Hoteles 
       (Nombre, Direccion, Ciudad, Pais, Telefono, EmailContacto, CategoriaEstrellas, DescripcionGeneral) 
       VALUES (?, ?, ?, ?, ?, ?, ?, ?)`,
      [
        Nombre,
        Direccion,
        Ciudad || null,
        Pais || null,
        Telefono || null,
        EmailContacto || null,
        CategoriaEstrellas || null,
        DescripcionGeneral || null,
      ]
    );

    const insertedId = result.insertId;
    const hoteles = await query('SELECT * FROM Hoteles WHERE HotelID = ?', [insertedId]);

    if (hoteles.length === 0) {
      return res.status(404).json({ error: 'Hotel no encontrado después de insertar' });
    }

    res.status(201).json({ message: 'Hotel creado exitosamente', hotel: hoteles[0] });
  } catch (error) {
    console.error('Error creando hotel:', error);
    res.status(500).json({ error: 'Error interno al crear hotel', details: error.message });
  }
});

// 3. Asociar servicio a hotel
app.post('/api/services/assign', async (req, res) => {
  try {
    const { HotelID, ServicioID } = req.body;

    if (!HotelID || !ServicioID) {
      return res.status(400).json({ error: 'HotelID y ServicioID son obligatorios' });
    }

    const hoteles = await query('SELECT * FROM Hoteles WHERE HotelID = ?', [HotelID]);
    if (hoteles.length === 0) {
      return res.status(404).json({ error: 'Hotel no encontrado' });
    }

    const servicios = await query('SELECT * FROM Servicios WHERE ServicioID = ?', [ServicioID]);
    if (servicios.length === 0) {
      return res.status(404).json({ error: 'Servicio no encontrado' });
    }

    const existing = await query(
      'SELECT * FROM HotelServicios WHERE HotelID = ? AND ServicioID = ?',
      [HotelID, ServicioID]
    );
    if (existing.length > 0) {
      return res.status(409).json({ error: 'El servicio ya está asociado al hotel' });
    }

    await query(
      'INSERT INTO HotelServicios (HotelID, ServicioID) VALUES (?, ?)',
      [HotelID, ServicioID]
    );

    res.status(201).json({ message: 'Servicio asociado al hotel exitosamente' });
  } catch (error) {
    console.error('Error asociando servicio al hotel:', error);
    res.status(500).json({ error: 'Error interno al asociar servicio al hotel', details: error.message });
  }
});

// 6. Buscar habitaciones disponibles
app.get('/api/habitaciones/disponibles', async (req, res) => {
  try {
    const { hotelId, capacidadMin, fechaInicio, fechaFin } = req.query;

    if (!hotelId || !capacidadMin || !fechaInicio || !fechaFin) {
      return res.status(400).json({ message: 'Faltan parámetros requeridos' });
    }

    const habitaciones = await query(`
      SELECT h.HabitacionID, h.NumeroHabitacion, t.NombreTipo, t.CapacidadMax, h.PrecioPorNoche, h.HotelID
      FROM Habitaciones h
      JOIN TiposHabitacion t ON h.TipoHabitacionID = t.TipoHabitacionID
      JOIN RoomServicios rs ON h.HabitacionID = rs.HabitacionID
      JOIN Servicios s ON rs.ServicioID = s.ServicioID
      WHERE s.NombreServicio = 'WiFi Gratis'
        AND h.HotelID = ?
        AND t.CapacidadMax >= ?
        AND h.HabitacionID NOT IN (
          SELECT r.HabitacionID
          FROM Reservas r
          WHERE r.EstadoReserva IN ('Pendiente', 'Confirmada')
            AND NOT (r.FechaCheckOut <= ? OR r.FechaCheckIn >= ?)
        )`,
      [hotelId, capacidadMin, fechaInicio, fechaFin]);

    res.json({ habitaciones });
  } catch (error) {
    console.error('Error buscando habitaciones disponibles:', error);
    res.status(500).json({ message: 'Error buscando habitaciones', error: error.message });
  }
});

// 9. Ingresos último mes
app.get('/api/hoteles/ingresos-ultimo-mes', async (req, res) => {
  try {
    const ingresos = await query(`
      SELECT
        h.HotelID,
        ho.Nombre AS NombreHotel,
        SUM(f.MontoTotal) AS TotalFacturadoPagado
      FROM Facturas f
      JOIN Reservas r ON f.ReservaID = r.ReservaID
      JOIN Habitaciones h ON r.HabitacionID = h.HabitacionID
      JOIN Hoteles ho ON h.HotelID = ho.HotelID
      WHERE f.EstadoPago = 'Pagado'
        AND f.FechaEmision >= DATE_SUB(CURDATE(), INTERVAL 1 MONTH)
      GROUP BY h.HotelID, ho.Nombre
    `);
    res.json({
      message: "Ingresos totales facturados y pagados por hotel en el último mes",
      ingresos
    });
  } catch (error) {
    console.error('Error obteniendo ingresos por hotel:', error);

// 11. Ocupación actual por hotel
app.get('/api/hoteles/ocupacion-actual', async (req, res) => {
  try {
    const ocupacion = await query(`
      SELECT 
        h.HotelID,
        h.Nombre AS NombreHotel,
        COUNT(DISTINCT hab.HabitacionID) AS TotalHabitaciones,
        COUNT(DISTINCT r.HabitacionID) AS HabitacionesOcupadas,
        ROUND(
          COUNT(DISTINCT r.HabitacionID) / COUNT(DISTINCT hab.HabitacionID) * 100, 
          2
        ) AS PorcentajeOcupacion
      FROM Hoteles h
      JOIN Habitaciones hab ON hab.HotelID = h.HotelID
      LEFT JOIN Reservas r ON r.HabitacionID = hab.HabitacionID
        AND CURDATE() BETWEEN r.FechaCheckIn AND r.FechaCheckOut
        AND r.EstadoReserva = 'Confirmada'
      GROUP BY h.HotelID, h.Nombre
    `);
    res.json({
      message: 'Ocupación actual por hotel',
      ocupacion
    });
  } catch (error) {
    console.error('Error obteniendo ocupación:', error);
    res.status(500).json({ message: 'Error al obtener ocupación', error: error.message });
  }
});

// 12. Crear reserva completa
app.post('/api/bookings/complete', async (req, res) => {
  const {
    HabitacionID,
    ClienteID,
    FechaCheckIn,
    FechaCheckOut,
    CodigoPromo = ''
  } = req.body;

  if (!HabitacionID || !ClienteID || !FechaCheckIn || !FechaCheckOut) {
    return res.status(400).json({ message: 'Faltan parámetros requeridos' });
  }

  try {
    const results = await query(
      `CALL CrearReserva(?, ?, ?, ?, ?, @ReservaID, @FacturaID, @Mensaje);
       SELECT @ReservaID AS ReservaID, @FacturaID AS FacturaID, @Mensaje AS Mensaje;`,
      [HabitacionID, ClienteID, FechaCheckIn, FechaCheckOut, CodigoPromo]
    );

    const output = results[1][0];

    if (!output.ReservaID) {
      return res.status(400).json({ message: output.Mensaje });
    }

    res.status(201).json({
      message: output.Mensaje,
      ReservaID: output.ReservaID,
      FacturaID: output.FacturaID
    });

  } catch (error) {
    console.error('Error creando reserva:', error);
    res.status(500).json({ message: 'Error interno al crear reserva', error: error.message });
  }
});
