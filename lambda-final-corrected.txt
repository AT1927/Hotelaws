const express = require('express');
const serverless = require('serverless-http');
const cors = require('cors');
const mysql = require('mysql2/promise');

const app = express();
app.use(cors());
app.use(express.json());

const pool = mysql.createPool({
  host: process.env.DB_HOST,
  user: process.env.DB_USER,
  password: process.env.DB_PASSWORD,
  database: process.env.DB_NAME,
  waitForConnections: true,
  connectionLimit: 10,
  ssl: { rejectUnauthorized: false },
  multipleStatements: true
});

const query = async (sql, params) => {
  const [rows] = await pool.execute(sql, params);
  return rows;
};

// 1. Registrar un nuevo Hotel
app.post('/api/hotels', async (req, res) => {
  try {
    const { Nombre, Direccion, Ciudad, Pais, Telefono, EmailContacto, CategoriaEstrellas, DescripcionGeneral } = req.body;

    if (!Nombre || !Direccion) {
      return res.status(400).json({ error: 'Nombre y Dirección son obligatorios' });
    }

    const result = await query(
      `INSERT INTO Hoteles 
       (Nombre, Direccion, Ciudad, Pais, Telefono, EmailContacto, CategoriaEstrellas, DescripcionGeneral) 
       VALUES (?, ?, ?, ?, ?, ?, ?, ?)`,
      [
        Nombre,
        Direccion,
        Ciudad || null,
        Pais || null,
        Telefono || null,
        EmailContacto || null,
        CategoriaEstrellas || null,
        DescripcionGeneral || null,
      ]
    );

    const insertedId = result.insertId;
    const hoteles = await query('SELECT * FROM Hoteles WHERE HotelID = ?', [insertedId]);

    if (hoteles.length === 0) {
      return res.status(404).json({ error: 'Hotel no encontrado después de insertar' });
    }

    res.status(201).json({ message: 'Hotel creado exitosamente', hotel: hoteles[0] });
  } catch (error) {
    console.error('Error creando hotel:', error);
    res.status(500).json({ error: 'Error interno al crear hotel', details: error.message });
  }
});

// 3. Asociar servicio a hotel
app.post('/api/services/assign', async (req, res) => {
  try {
    const { HotelID, ServicioID } = req.body;

    if (!HotelID || !ServicioID) {
      return res.status(400).json({ error: 'HotelID y ServicioID son obligatorios' });
    }

    const hoteles = await query('SELECT * FROM Hoteles WHERE HotelID = ?', [HotelID]);
    if (hoteles.length === 0) {
      return res.status(404).json({ error: 'Hotel no encontrado' });
    }

    const servicios = await query('SELECT * FROM Servicios WHERE ServicioID = ?', [ServicioID]);
    if (servicios.length === 0) {
      return res.status(404).json({ error: 'Servicio no encontrado' });
    }

    const existing = await query(
      'SELECT * FROM HotelServicios WHERE HotelID = ? AND ServicioID = ?',
      [HotelID, ServicioID]
    );
    if (existing.length > 0) {
      return res.status(409).json({ error: 'El servicio ya está asociado al hotel' });
    }

    await query(
      'INSERT INTO HotelServicios (HotelID, ServicioID) VALUES (?, ?)',
      [HotelID, ServicioID]
    );

    res.status(201).json({ message: 'Servicio asociado al hotel exitosamente' });
  } catch (error) {
    console.error('Error asociando servicio al hotel:', error);
    res.status(500).json({ error: 'Error interno al asociar servicio al hotel', details: error.message });
  }
});

// 5.1. Crear nueva habitación
app.post('/api/rooms/create', async (req, res) => {
  try {
    const {
      HotelID,
      TipoHabitacionID,
      NumeroHabitacion,
      PrecioPorNoche,
      Estado = 'Disponible'
    } = req.body;

    if (!HotelID || !TipoHabitacionID || !NumeroHabitacion || !PrecioPorNoche) {
      return res.status(400).json({ error: 'HotelID, TipoHabitacionID, NumeroHabitacion y PrecioPorNoche son obligatorios' });
    }

    // Validar que el hotel existe
    const hotel = await query('SELECT * FROM Hoteles WHERE HotelID = ?', [HotelID]);
    if (hotel.length === 0) {
      return res.status(404).json({ error: 'Hotel no encontrado' });
    }

    // Validar que el tipo de habitación existe
    const tipoHabitacion = await query('SELECT * FROM TiposHabitacion WHERE TipoHabitacionID = ?', [TipoHabitacionID]);
    if (tipoHabitacion.length === 0) {
      return res.status(404).json({ error: 'Tipo de habitación no encontrado' });
    }

    // Validar que el número de habitación no exista en el hotel
    const habitacionExistente = await query(
      'SELECT * FROM Habitaciones WHERE HotelID = ? AND NumeroHabitacion = ?',
      [HotelID, NumeroHabitacion]
    );
    if (habitacionExistente.length > 0) {
      return res.status(409).json({ error: 'Ya existe una habitación con este número en el hotel' });
    }

    const result = await query(`
      INSERT INTO Habitaciones (HotelID, TipoHabitacionID, NumeroHabitacion, PrecioPorNoche, Estado)
      VALUES (?, ?, ?, ?, ?)
    `, [HotelID, TipoHabitacionID, NumeroHabitacion, PrecioPorNoche, Estado]);

    const habitacionCreada = await query(`
      SELECT h.*, ho.Nombre as NombreHotel, t.NombreTipo, t.CapacidadMax
      FROM Habitaciones h
      LEFT JOIN Hoteles ho ON h.HotelID = ho.HotelID
      LEFT JOIN TiposHabitacion t ON h.TipoHabitacionID = t.TipoHabitacionID
      WHERE h.HabitacionID = ?
    `, [result.insertId]);

    res.status(201).json({
      message: 'Habitación creada exitosamente',
      habitacion: habitacionCreada[0]
    });

  } catch (error) {
    console.error('Error creando habitación:', error);
    res.status(500).json({ error: 'Error interno al crear habitación', details: error.message });
  }
});

app.get('/api/habitaciones/disponibles', async (req, res) => {
  try {
    const { hotelId, capacidadMin, fechaInicio, fechaFin, wifi } = req.query;

    if (!hotelId || !capacidadMin || !fechaInicio || !fechaFin) {
      return res.status(400).json({ message: 'Faltan parámetros requeridos' });
    }

    let wifiCondition = '';
    if (wifi === 'true') {
      wifiCondition = `
        AND EXISTS (
          SELECT 1 FROM RoomServicios rs 
          JOIN Servicios s ON rs.ServicioID = s.ServicioID 
          WHERE rs.HabitacionID = h.HabitacionID 
          AND s.NombreServicio = 'WiFi Gratis'
        )`;
    }

    const habitaciones = await query(`
      SELECT h.HabitacionID, h.NumeroHabitacion, t.NombreTipo, t.CapacidadMax, h.PrecioPorNoche, h.HotelID
      FROM Habitaciones h
      JOIN TiposHabitacion t ON h.TipoHabitacionID = t.TipoHabitacionID
      WHERE h.HotelID = ?
        AND t.CapacidadMax >= ?
        AND h.Estado = 'Disponible'
        ${wifiCondition}
        AND h.HabitacionID NOT IN (
          SELECT r.HabitacionID
          FROM Reservas r
          WHERE r.EstadoReserva IN ('Pendiente', 'Confirmada')
            AND NOT (r.FechaCheckOut <= ? OR r.FechaCheckIn >= ?)
        )`,
      [hotelId, capacidadMin, fechaInicio, fechaFin]);

    res.json({ habitaciones });
  } catch (error) {
    console.error('Error buscando habitaciones disponibles:', error);
    res.status(500).json({ message: 'Error buscando habitaciones', error: error.message });
  }
});

// 9. Ingresos por hotel con filtros mejorados
app.get('/api/hoteles/ingresos-ultimo-mes', async (req, res) => {
  try {
    const { hotelId, fechaInicio, fechaFin } = req.query;
    
    let whereCondition = "WHERE f.EstadoPago = 'Pagado'";
    let params = [];
    
    if (fechaInicio && fechaFin) {
      whereCondition += " AND f.FechaEmision BETWEEN ? AND ?";
      params.push(fechaInicio, fechaFin);
    } else {
      whereCondition += " AND f.FechaEmision >= DATE_SUB(CURDATE(), INTERVAL 1 MONTH)";
    }
    
    if (hotelId) {
      whereCondition += " AND h.HotelID = ?";
      params.push(hotelId);
    }

    const ingresos = await query(`
      SELECT
        h.HotelID,
        ho.Nombre AS NombreHotel,
        COALESCE(SUM(f.MontoTotal), 0) AS TotalFacturadoPagado,
        COUNT(f.FacturaID) AS TotalFacturas,
        COALESCE(AVG(f.MontoTotal), 0) AS PromedioFactura
      FROM Hoteles ho
      LEFT JOIN Habitaciones h ON h.HotelID = ho.HotelID
      LEFT JOIN Reservas r ON r.HabitacionID = h.HabitacionID
      LEFT JOIN Facturas f ON f.ReservaID = r.ReservaID AND f.EstadoPago = 'Pagado'
      ${fechaInicio && fechaFin ? 'AND f.FechaEmision BETWEEN ? AND ?' : 'AND (f.FechaEmision IS NULL OR f.FechaEmision >= DATE_SUB(CURDATE(), INTERVAL 1 MONTH))'}
      ${hotelId ? 'WHERE ho.HotelID = ?' : ''}
      GROUP BY h.HotelID, ho.Nombre
      ORDER BY TotalFacturadoPagado DESC
    `, params);
    
    res.json({
      message: "Ingresos totales facturados y pagados por hotel",
      ingresos
    });
  } catch (error) {
    console.error('Error obteniendo ingresos por hotel:', error);
    res.status(500).json({ message: 'Error al obtener ingresos por hotel', error: error.message });
  }
});

// 11. Ocupación por hotel con filtros mejorados
app.get('/api/hoteles/ocupacion-actual', async (req, res) => {
  try {
    const { hotelId, fecha } = req.query;
    
    let whereCondition = "";
    let params = [];
    let fechaCondition = "CURDATE()";
    
    if (fecha) {
      fechaCondition = "?";
      params.push(fecha);
    }
    
    if (hotelId) {
      whereCondition = "WHERE h.HotelID = ?";
      params.push(hotelId);
    }

    const ocupacion = await query(`
      SELECT 
        h.HotelID,
        h.Nombre AS NombreHotel,
        COUNT(DISTINCT hab.HabitacionID) AS TotalHabitaciones,
        COUNT(DISTINCT CASE 
          WHEN r.ReservaID IS NOT NULL 
          AND ${fechaCondition} BETWEEN r.FechaCheckIn AND r.FechaCheckOut
          AND r.EstadoReserva IN ('Confirmada', 'Pendiente') 
          THEN r.HabitacionID 
        END) AS HabitacionesOcupadas,
        ROUND(
          COALESCE(COUNT(DISTINCT CASE 
            WHEN r.ReservaID IS NOT NULL 
            AND ${fechaCondition} BETWEEN r.FechaCheckIn AND r.FechaCheckOut
            AND r.EstadoReserva IN ('Confirmada', 'Pendiente') 
            THEN r.HabitacionID 
          END) / NULLIF(COUNT(DISTINCT hab.HabitacionID), 0) * 100, 0), 
          2
        ) AS PorcentajeOcupacion
      FROM Hoteles h
      LEFT JOIN Habitaciones hab ON hab.HotelID = h.HotelID
      LEFT JOIN Reservas r ON r.HabitacionID = hab.HabitacionID
      ${whereCondition}
      GROUP BY h.HotelID, h.Nombre
      ORDER BY PorcentajeOcupacion DESC
    `, params);
    
    res.json({
      message: 'Ocupación por hotel',
      ocupacion
    });
  } catch (error) {
    console.error('Error obteniendo ocupación:', error);
    res.status(500).json({ message: 'Error al obtener ocupación', error: error.message });
  }
});

        )
    `, [HabitacionID, FechaCheckIn, FechaCheckOut]);

    if (disponibilidad.length === 0) {
      return res.status(400).json({ message: 'La habitación no está disponible para las fechas seleccionadas' });
    }

    const habitacion = disponibilidad[0];
    const diasEstancia = Math.ceil((new Date(FechaCheckOut) - new Date(FechaCheckIn)) / (1000 * 60 * 60 * 24));
    let costoTotal = habitacion.PrecioPorNoche * diasEstancia;
    let descuentoAplicado = 0;
    let promocionID = null;

    // Validar y aplicar código promocional si se proporciona
    if (CodigoPromo) {
      const promocion = await query(`
        SELECT PromocionID, DescuentoPorcentaje, DescuentoFijo
        FROM Promociones 
        WHERE CodigoPromo = ? 
          AND Activa = TRUE 
          AND CURDATE() BETWEEN FechaInicio AND FechaFin
      `, [CodigoPromo]);

      if (promocion.length > 0) {
        const promo = promocion[0];
        promocionID = promo.PromocionID;
        
        if (promo.DescuentoPorcentaje) {
          descuentoAplicado = costoTotal * (promo.DescuentoPorcentaje / 100);
        } else if (promo.DescuentoFijo) {
          descuentoAplicado = Math.min(promo.DescuentoFijo, costoTotal);
        }
        
        costoTotal -= descuentoAplicado;
      }
    }

    // Crear la reserva
    const reservaResult = await query(`
      INSERT INTO Reservas (HabitacionID, ClienteID, FechaReserva, FechaCheckIn, FechaCheckOut, EstadoReserva, CostoTotal)
      VALUES (?, ?, CURDATE(), ?, ?, 'Pendiente', ?)
    `, [HabitacionID, ClienteID, FechaCheckIn, FechaCheckOut, costoTotal]);

    const reservaID = reservaResult.insertId;

    // Asociar promoción si se aplicó
    if (promocionID) {
      await query(`
        INSERT INTO ReservaPromociones (ReservaID, PromocionID, DescuentoAplicado)
        VALUES (?, ?, ?)
      `, [reservaID, promocionID, descuentoAplicado]);
    }

    // Crear factura inicial
    const facturaResult = await query(`
      INSERT INTO Facturas (ReservaID, FechaEmision, MontoTotal, EstadoPago, MetodoPago)
      VALUES (?, CURDATE(), ?, 'Pendiente', 'Efectivo')
    `, [reservaID, costoTotal]);

    const facturaID = facturaResult.insertId;

    res.status(201).json({
      message: 'Reserva creada exitosamente',
      ReservaID: reservaID,
      FacturaID: facturaID,
      CostoTotal: costoTotal,
      DescuentoAplicado: descuentoAplicado,
      DiasEstancia: diasEstancia,
      HotelNombre: habitacion.NombreHotel,
      TipoHabitacion: habitacion.NombreTipo
    });

  } catch (error) {
    console.error('Error creando reserva:', error);
    res.status(500).json({ message: 'Error interno al crear reserva', error: error.message });
  }
});

// Crear trigger para actualizar puntuación media automáticamente
app.post('/api/database/create-trigger', async (req, res) => {
  try {
    const triggerSQL = `
      DROP TRIGGER IF EXISTS tr_actualizar_puntuacion_hotel;
      
      CREATE TRIGGER tr_actualizar_puntuacion_hotel
      AFTER INSERT ON Reseñas
      FOR EACH ROW
      BEGIN
          DECLARE hotel_id INT;
          
          -- Obtener el HotelID de la reserva
          SELECT h.HotelID INTO hotel_id
          FROM Reservas r
          JOIN Habitaciones hab ON r.HabitacionID = hab.HabitacionID
          JOIN Hoteles h ON hab.HotelID = h.HotelID
          WHERE r.ReservaID = NEW.ReservaID;
          
          -- Actualizar la puntuación media del hotel
          UPDATE Hoteles 
          SET PuntuacionMedia = (
              SELECT COALESCE(AVG(res.Puntuacion), 0)
              FROM Reseñas res
              JOIN Reservas r ON res.ReservaID = r.ReservaID
              JOIN Habitaciones hab ON r.HabitacionID = hab.HabitacionID
              WHERE hab.HotelID = hotel_id AND res.Visible = TRUE
          )
          WHERE HotelID = hotel_id;
      END;
      
      DROP TRIGGER IF EXISTS tr_actualizar_puntuacion_hotel_update;
      
      CREATE TRIGGER tr_actualizar_puntuacion_hotel_update
      AFTER UPDATE ON Reseñas
      FOR EACH ROW
      BEGIN
          DECLARE hotel_id INT;
          
          -- Obtener el HotelID de la reserva
          SELECT h.HotelID INTO hotel_id
          FROM Reservas r
          JOIN Habitaciones hab ON r.HabitacionID = hab.HabitacionID
          JOIN Hoteles h ON hab.HotelID = h.HotelID
          WHERE r.ReservaID = NEW.ReservaID;
          
          -- Actualizar la puntuación media del hotel
          UPDATE Hoteles 
          SET PuntuacionMedia = (
              SELECT COALESCE(AVG(res.Puntuacion), 0)
              FROM Reseñas res
              JOIN Reservas r ON res.ReservaID = r.ReservaID
              JOIN Habitaciones hab ON r.HabitacionID = hab.HabitacionID
              WHERE hab.HotelID = hotel_id AND res.Visible = TRUE
          )
          WHERE HotelID = hotel_id;
      END;
    `;

    await query(triggerSQL);
    
    res.json({
      message: 'Triggers para actualización automática de puntuación media creados exitosamente'
    });
  } catch (error) {
    console.error('Error creando triggers:', error);
    res.status(500).json({ message: 'Error creando triggers', error: error.message });
  }
});
